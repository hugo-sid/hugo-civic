{{- /* A single accordion panel. Must be nested inside {{< accordion >}} — it reads heading_level from .Parent so the whole group keeps one consistent heading level in the document outline. aria-controls and the panel id are derived from .Ordinal, which is unique per shortcode name per page, so ids stay stable and collision-free across multiple accordions on one page. */ -}} {{- $h := "h4" -}} {{- with .Parent }}{{ $h = .Get "heading_level" | default "h4" }}{{ end -}} {{- $id := printf "accordion-%d" .Ordinal -}} {{- /* Heading tags are built as strings and marked safeHTML: Go's html/template cannot parse a dynamic tag name, and `<{{ $h }}>` is emitted escaped as `<h4>` rather than as an element. */ -}} {{- $hOpen := printf "<%s class=\"usa-accordion__heading\">" $h | safeHTML -}} {{- $hClose := printf "" $h | safeHTML -}} {{ $hOpen }} {{ $hClose }}
{{ .Inner | .Page.RenderString (dict "display" "block") }}
